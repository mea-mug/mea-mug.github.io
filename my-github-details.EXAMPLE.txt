#  WHAT TO DO
#
#   1. Make a copy of this file, in this same folder.
#      (Right-click this file -> Copy, then right-click the empty
#       space in the folder -> Paste)
#
#   2. Rename the copy to exactly:
#
#         my-github-details.txt
#
#      That is this file's name with ".EXAMPLE" taken out.
#      If Windows hides file endings, this file shows up as
#      "my-github-details.EXAMPLE" - in that case rename the copy to
#      just "my-github-details" and Windows keeps the .txt for you.
#
#   3. Open your copy (double-click it, it opens in Notepad) and
#      replace each PASTE_..._HERE below with your own value.
#
#   4. Save the file and close it.
#
#  THREE RULES FOR THE LINES BELOW
#   * Start the line with the name - no spaces in front of it.
#   * No spaces around the = sign.    GOOD: NAME=value
#                                     BAD:  NAME = value
#   * Nothing after your value - no trailing spaces.
#
#  GOOD TO KNOW
#   * Lines starting with # are notes for you and are ignored.
#     Blank lines are ignored too.
#   * Your copy (my-github-details.txt) is never uploaded to GitHub.
#     It is listed in .gitignore, and that is what keeps it on your PC.
#   * Never e-mail this file, never put it in a ZIP you send to someone,
#     and never post a screenshot of it. The token at the bottom is a
#     password to your GitHub account. If it ever leaks, delete it at
#     https://github.com/settings/tokens and make a new one.
#
# ---------------------------------------------------------------------


# Your GitHub account name. It is the last part of your profile address:
# https://github.com/YOUR-ACCOUNT-NAME-IS-HERE
GITHUB_USERNAME=PASTE_YOUR_GITHUB_USERNAME_HERE


# The e-mail address of your GitHub account. GitHub stamps this on every
# change you publish, so use the one you signed up with.
GITHUB_EMAIL=PASTE_YOUR_GITHUB_EMAIL_HERE


# The name of the repository that holds your website. For a free personal
# site this MUST be your account name followed by .github.io
# For example, if your account name is jane, it is:  jane.github.io
GITHUB_REPO=PASTE_YOUR_REPOSITORY_NAME_HERE


# Your Personal Access Token. This is a password - treat it like one.
# Make one at:  https://github.com/settings/tokens
#   -> "Generate new token" -> "Generate new token (classic)"
#   -> Note: anything you like, for example "my website"
#   -> Expiration: "No expiration"
#   -> Tick the box "repo" and the box "workflow"
#   -> "Generate token", then copy the line that starts with ghp_
# GitHub only shows it once, so paste it here straight away.
GITHUB_TOKEN=PASTE_YOUR_TOKEN_HERE
